## 执行前设定（对应需求文档第 8 节）

| 决策 | 取值 | 理由 |
| --- | --- | --- |
| 非规范语言路径 | 保留 + `noindex` + canonical 指向规范页 | GitHub Pages 无法 301，硬删会让已有外链 404 |
| Mermaid | 客户端按需动态 import，本地打包 | 构建期渲染需 Playwright/Chromium，CI 成本高于收益；仍是零 CDN、断网可用 |
| 论文笔记范围 | 只写已发表论文 | 避免未发表数据与企业数据合规问题 |
| 更新节奏 | 轻量脚本 + 模板骨架 | 先降低启动摩擦，不预设产出频率 |

```ts
export interface VisibleOptions {
  /** 仅本地开发使用；构建时始终为 false。 */
  includeDrafts?: boolean;
}

function devBuild(): boolean {
  try {
    return Boolean(import.meta.env?.DEV);
  } catch {
    return false;
  }
}

export function visiblePosts<T extends Post>(
  posts: T[],
  today = localToday(),
  options: VisibleOptions = {}
): T[] {
  const includeDrafts = options.includeDrafts ?? devBuild();
  return posts
    .filter((post) => (includeDrafts || !post.data.draft) && post.data.date <= today)
    .sort((a, b) => b.data.date.localeCompare(a.data.date) || a.id.localeCompare(b.id));
}
```

`scripts/check-posts.mjs` 用 `tsx` 运行，复用 `src/lib/schema.ts`：

```ts
import { readdir, readFile } from 'node:fs/promises';
import { parse } from 'yaml';
import { blogSchema } from '../src/lib/schema.ts';

const errors: string[] = [];
const files = (await readdir('src/content/blog')).filter((f) => /\.mdx?$/.test(f));

for (const file of files) {
  const slug = file.replace(/\.mdx?$/, '');
  const source = await readFile(`src/content/blog/${file}`, 'utf8');
  const match = source.match(/^---\n([\s\S]*?)\n---/);
  if (!match) { errors.push(`${file}: missing frontmatter`); continue; }
  try {
    const data = blogSchema.parse(parse(match[1]!));
    if (!/^[a-z0-9]+(?:-[a-z0-9]+)*$/.test(slug)) errors.push(`${file}: slug must be kebab-case`);
    if (data.updated && data.updated < data.date) errors.push(`${file}: updated is earlier than date`);
    if (data.summary.startsWith('TODO')) errors.push(`${file}: replace the placeholder summary`);
  } catch (error) {
    errors.push(`${file}: ${error instanceof Error ? error.message : String(error)}`);
  }
}

if (errors.length) { console.error(errors.join('\n')); process.exit(1); }
console.log(`checked ${files.length} post(s)`);
```

// src/lib/posts.ts
import type { Lang } from '../i18n/ui';

export function postsByLang<T extends Post>(posts: T[], lang: Lang): T[] {
  return posts.filter((post) => post.data.lang === lang);
}

export function canonicalPostPath(post: Post): string {
  return post.data.lang === 'zh' ? `zh/blog/${post.id}` : `blog/${post.id}`;
}

export function canonicalPostUrl(
  post: Post,
  base = import.meta.env.BASE_URL.replace(/\/$/, '')
): string {
  return `${base}/${canonicalPostPath(post)}/`;
}

export function isCanonicalPath(post: Post, pathname: string): boolean {
  return pathname.startsWith(canonicalPostUrl(post).replace(/\/$/, ''));
}

export function translationFor<T extends Post>(posts: T[], post: T): T | undefined {
  const explicit = post.data.translationOf
    ? posts.find((item) => item.id === post.data.translationOf && item.data.lang !== post.data.lang)
    : undefined;
  if (explicit) return explicit;
  return posts.find(
    (item) => item.data.translationOf === post.id && item.data.lang !== post.data.lang
  );
}
```

// src/lib/publications.ts
export function findPublication<T extends Pub>(items: T[], ref: string): T | undefined {
  return items.find(
    (pub) => pub.id === ref || pub.data.doi === ref || (pub.data.doi && `https://doi.org/${pub.data.doi}` === ref)
  );
}

export function notesByPublication<T extends Pub, P extends { data: { paper?: string } }>(
  posts: P[],
  pubs: T[]
): Map<string, P[]> {
  const map = new Map<string, P[]>();
  for (const post of posts) {
    if (!post.data.paper) continue;
    const pub = findPublication(pubs, post.data.paper);
    if (!pub) continue;
    map.set(pub.id, [...(map.get(pub.id) ?? []), post]);
  }
  return map;
}

export function sortPaperReading<T extends { data: { paper?: string; date: string } }>(
  posts: T[],
  pubs: Pub[]
): T[] {
  return [...posts].sort((a, b) => {
    const ya = findPublication(pubs, a.data.paper ?? '')?.data.year ?? 0;
    const yb = findPublication(pubs, b.data.paper ?? '')?.data.year ?? 0;
    return yb - ya || b.data.date.localeCompare(a.data.date);
  });
}
```

```ts
// src/lib/relations.ts
export function validateBlogPapers(
  posts: { id: string; data: { paper?: string } }[],
  publications: Pub[]
): string[] {
  const errors: string[] = [];
  for (const post of posts) {
    if (post.data.paper && !findPublication(publications, post.data.paper)) {
      errors.push(`blog/${post.id}.md: paper "${post.data.paper}" not found`);
    }
  }
  return errors;
}

```ts
// src/lib/markdown.ts
export interface CodeMeta {
  title?: string;
  highlight: [number, number][];
}

export function parseCodeMeta(meta: string): CodeMeta {
  const title = meta.match(/title="([^"]+)"/)?.[1];
  const ranges = meta.match(/\{([^}]+)\}/)?.[1];
  const highlight: [number, number][] = [];
  for (const part of ranges?.split(',') ?? []) {
    const [start, end] = part.trim().split('-');
    const from = Number.parseInt(start ?? '', 10);
    if (Number.isNaN(from)) continue;
    const to = Number.parseInt(end ?? start ?? '', 10);
    highlight.push([from, Number.isNaN(to) ? from : to]);
  }
  return { title, highlight };
}

export function hasMermaid(body: string): boolean {
  return /(^|\n)```mermaid(\s|$)/.test(body);
}
```

`scripts/shiki-code-meta.mjs`：

```js
import { parseCodeMeta } from '../src/lib/markdown.ts';

export function codeMetaTransformer() {
  return {
    name: 'code-meta',
    pre(node) {
      const raw = this.options?.meta?.__raw ?? '';
      const { title } = parseCodeMeta(raw);
      if (title) node.properties['data-code-title'] = title;
    },
    line(node, line) {
      const raw = this.options?.meta?.__raw ?? '';
      const { highlight } = parseCodeMeta(raw);
      if (highlight.some(([from, to]) => line >= from && line <= to)) {
        node.properties.class = `${node.properties.class ?? ''} highlighted`;
      }
    },
  };
}

## 偏离与风险

1. **Mermaid 走客户端按需加载**，与需求文档“优先构建期”不同；理由是避免 CI 安装 Chromium，且仍是本地资源、断网可用。
2. **不提交 git**，与 skill 默认“每任务 commit”不同；按工具约束需用户显式要求才提交。
3. **Shiki transformer 的 meta 访问方式**依赖 Astro 内置 Shiki 版本，若 `this.options.meta.__raw` 不可用，步骤 4.4 已给出回退方案。
4. **现有唯一一篇中文文章**在实施后会新增 `/zh/blog/...` 规范路径，英文路径变 noindex；原 `/blog/...` 链接仍可访问。
